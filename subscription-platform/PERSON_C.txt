### Day 2: Deploy Notification Lambda

Lambda is an AWS service that runs your code without servers.

FILE:

ACTION:
Verify email templates

REASON:
Contains the text for the 3 types of emails sent to users.

CODE:
function paymentSuccess(email) {
  return { subject: "Subscription Activated", body: `Hello ${email},\nYour subscription is active.` };
}
function paymentFailed(email) {
  return { subject: "Payment Failed", body: `Hello ${email},\nPayment failed.` };
}
function subscriptionExpired(email) {
  return { subject: "Subscription Expired", body: `Hello,\nYour subscription expired.` };
}
module.exports = { paymentSuccess, paymentFailed, subscriptionExpired };

FILE:
backend/notification-lambda/index.js

ACTION:
Verify notification listener logic

REASON:
Listens to SNS and sends emails via SES.

CODE:
const { SESClient, SendEmailCommand } = require("@aws-sdk/client-ses");
const { paymentSuccess, paymentFailed, subscriptionExpired } = require("./emailTemplates");
const sesClient = new SESClient({ region: process.env.AWS_REGION || "ap-southeast-1" });

exports.handler = async (event) => {
  for (const record of event.Records) {
    const payload = JSON.parse(record.Sns.Message);
    let template;
    if (payload.type === "PAYMENT_SUCCESS") template = paymentSuccess(payload.email);
    else if (payload.type === "PAYMENT_FAILED") template = paymentFailed(payload.email);
    else if (payload.type === "SUBSCRIPTION_EXPIRED") template = subscriptionExpired(payload.email);
    else continue;

    await sesClient.send(new SendEmailCommand({
      Source: process.env.SENDER_EMAIL,
      Destination: { ToAddresses: [payload.email] },
      Message: { Subject: { Data: template.subject }, Body: { Text: { Data: template.body } } }
    }));
  }
  return { success: true };
};
